def draw_streamplot(ax, X, Y, U, V, x_range, y_range):
    valid = (~np.isnan(U) & ~np.isnan(V) & ~np.isnan(X) & ~np.isnan(Y))
    X, Y, U, V = X[valid], Y[valid], U[valid], V[valid]
    
    x_lin = np.linspace(x_range[0], x_range[1], 150)
    y_lin = np.linspace(y_range[0], y_range[1], 150)
    grid_x, grid_y = np.meshgrid(x_lin, y_lin)
    
    grid_u = np.nan_to_num(griddata((X, Y), U, (grid_x, grid_y), method='linear', fill_value=0))
    grid_v = np.nan_to_num(griddata((X, Y), V, (grid_x, grid_y), method='linear', fill_value=0))
    
    ax.streamplot(x_lin, y_lin, grid_u, grid_v, 
                  color='black', linewidth=1.0, arrowsize=1.0, density=1.2)
